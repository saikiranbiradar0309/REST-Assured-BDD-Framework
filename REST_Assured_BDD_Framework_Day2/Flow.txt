package spec;



import io.restassured.builder.RequestSpecBuilder;
import io.restassured.builder.ResponseSpecBuilder;
import io.restassured.filter.log.LogDetail;
import io.restassured.http.ContentType;
import io.restassured.specification.RequestSpecification;
import io.restassured.specification.ResponseSpecification;

public class SpecFactory {
	
	public static RequestSpecification requestSpec()
	{
		
		// RequestSpecBuilder is a class which is used to build the request specification
		RequestSpecBuilder spec = new RequestSpecBuilder();
		
		
		//Generally we store BaseUri in a property file and read it from there. 
		//But for now we will hardcode it.
		spec.setBaseUri("http://localhost:9191");
		
		//We can set the content type of the request here.
		spec.setContentType(ContentType.JSON);
		
		//We can set the accept type of the request here.
		spec.setAccept(ContentType.JSON);
		
		//We can set the relaxed HTTPS validation here.
		spec.setRelaxedHTTPSValidation();
		
		//We can log the request and response details here.
		spec.log(LogDetail.ALL);
		
		
		// I want to combine all the above details and create a request specification object which can be used in all the test cases.
		// So we will use build() method to create a request specification object.
		// We can use this request specification object in all the test cases.
		RequestSpecification requestSpec = spec.build();
		
		
		
		return requestSpec;
	}
	
	
	public static ResponseSpecification responseSpec()
	{
		
		// ResponseSpecBuilder is a class which is used to build the response specification
		ResponseSpecBuilder spec = new ResponseSpecBuilder();
		
		//We can set the expected Response Time of the response here.
		//spec.expectResponseTime(Matchers.lessThan(5000L));
		
		
		//spec.expectHeader("Content-Type", Matchers.containsString("application/json"));
		
		// log the request and response details here.
		spec.log(LogDetail.ALL);
		
		ResponseSpecification responseSpec =  spec.build();
		
		return responseSpec;
	}
	
}

		--> Also explain why we are taking pathParam and queryParam here
		--> WHats the usecase of  pathParam and queryParam in the client layer


package client;

import java.util.HashMap;

import io.restassured.RestAssured;
import io.restassured.response.Response;
import spec.SpecFactory;

public class BaseClient {
	
	
	public Response get(String endpoint)
	{
		
		Response response =  RestAssured
			.given()
				.spec(SpecFactory.requestSpec())
			.when()
				.get(endpoint)
			.then()
				.assertThat()
				.spec(SpecFactory.responseSpec())
				.extract()
				.response();
		
		
		
		return response;
	}
	
	
	public Response get(String endpoint, HashMap<String, String> pathParams)
	{
		
		Response response = RestAssured
			.given()
				.spec(SpecFactory.requestSpec())
				.pathParams(pathParams)
			.when()
				.get(endpoint)
			.then()
				.assertThat()
				.spec(SpecFactory.responseSpec())
				.extract()
				.response();
		
		return response;
	}
	
	
	
	
	public Response get(String endpoint, HashMap<String, String> pathParams, HashMap<String, String> queryParams)
	{
		
		Response response = null;
		if(pathParams != null)
		{
		
			 response = RestAssured
					.given()
						.spec(SpecFactory.requestSpec())
						.pathParams(pathParams)
					.when()
						.get(endpoint)
					.then()
						.assertThat()
						.spec(SpecFactory.responseSpec())
						.extract()
						.response();
		}
		
		if(queryParams != null)
		{
			response = RestAssured
					.given()
						.spec(SpecFactory.requestSpec())
						.queryParams(queryParams)
					.when()
						.get(endpoint)
					.then()
						.assertThat()
						.spec(SpecFactory.responseSpec())
						.extract()
						.response();
		}
		
		
		if(pathParams != null && queryParams != null)
		{
			response = RestAssured
					.given()
						.spec(SpecFactory.requestSpec())
						.pathParams(pathParams)
						.queryParams(queryParams)
					.when()
						.get(endpoint)
					.then()
						.assertThat()
						.spec(SpecFactory.responseSpec())
						.extract()
						.response();
		}
		
		return response;
	}
	
}

05. Now we create services layer package inside src/main/java 
		--> Inside services layer package create a EmployeeServices class



To check whether the request will Accept pathParam or queryParam, Goto the swagger UI,
		Inside Parameters section, we can see the type of parameter is mentioned as path or query.
		
		

package services;

import java.util.HashMap;

import client.BaseClient;
import endpoints.EmployeeEndpoint;
import io.restassured.response.Response;

public class EmployeeServices extends BaseClient {
	
	
	public Response getAllEmployees()
	{
		return this.get(EmployeeEndpoint.GET_ALL_EMPLOYEES);
	}
	
	
	public Response getAllEmployeeAsPerId(String id)
	{
		// So here we need to pass the ID as path parameter, 
		//so we need to create a HashMap and pass the ID as key value pair in the HashMap and then pass that HashMap to the get method of BaseClient class.
	
		HashMap<String, String> pathParams = new HashMap<String, String>();
		pathParams.put("id", id);
		
		
		return this.get(EmployeeEndpoint.GET_EMPLOYEE_AS_PER_ID, pathParams);
	
	}
	
	
	public Response getALlEmployeesAsPerCity(String city)
	{
		HashMap<String, String> pathParams = new HashMap<String, String>();
		pathParams.put("city", city);
		
		return this.get(EmployeeEndpoint.GET_EMPLOYEES_AS_PER_CITY, pathParams);
	}
}

package validators;

import org.hamcrest.Matchers;
import org.testng.Assert;

import io.restassured.response.Response;

public class Assertions {
	
	
	public static void verifyStatusCode(Response response, int expectedStatusCode)
	{
		//response.then().assertThat().statusCode(Matchers.equalTo(expectedStatusCode));
		
		
		Assert.assertEquals(response.getStatusCode(), expectedStatusCode);
	}
	
	
	public static void verifyResponseStatusLine(Response response, String expectedStatusLine)
	{
		Assert.assertTrue(response.statusLine().contains(expectedStatusLine));
	}
	
	
	public static void verifyResponseTimeShouldBelow(Response response, long expectedTime)
	{
		Assert.assertTrue(response.getTime() < expectedTime);
		
		response.then().assertThat().time(Matchers.lessThan(expectedTime));
	}
	
	
	public static void verifyNotNull(Response response, String jsonPath)
	{
		String actualValue = response.jsonPath().getString(jsonPath);
		
		Assert.assertNotNull(actualValue);
		
		
		//response.then().assertThat().body(jsonPath, Matchers.notNullValue());
	}
	
	
	
	public static void verifyInstanceOfValue(Response response, String jsonPath, Class<?> classType)
	{
		String actualValue = response.jsonPath().getString(jsonPath);
		Assert.assertEquals(actualValue.getClass(), classType);
		
		response.then().assertThat().body(jsonPath, Matchers.instanceOf(classType));
	}
	
}
